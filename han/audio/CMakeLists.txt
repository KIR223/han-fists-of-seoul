han_module(audio DEPENDS han_platform)

# miniaudio приходит из vcpkg (функция audio): либо CMake-пакет, либо просто заголовок miniaudio.h.
# Без него модуль собирается без звукового движка (hasAudio() == false), таблица событий работает.
find_package(miniaudio CONFIG QUIET)
set(han_audio_have_ma OFF)
if(miniaudio_FOUND)
    target_link_libraries(han_audio PUBLIC miniaudio::miniaudio)
    set(han_audio_have_ma ON)
else()
    find_path(MINIAUDIO_INCLUDE_DIRS "miniaudio.h")
    if(MINIAUDIO_INCLUDE_DIRS)
        target_include_directories(han_audio PRIVATE ${MINIAUDIO_INCLUDE_DIRS})
        set(han_audio_have_ma ON)
    endif()
endif()

get_target_property(han_audio_sources han_audio SOURCES)
if(han_audio_have_ma)
    list(FILTER han_audio_sources EXCLUDE REGEX "no_audio\\.cpp$")
    target_compile_definitions(han_audio PUBLIC HAN_HAS_MINIAUDIO=1)
    if(WIN32)
        target_link_libraries(han_audio PRIVATE ole32 winmm)
    endif()
else()
    list(FILTER han_audio_sources EXCLUDE REGEX "audio_system_ma\\.cpp$")
    message(STATUS "miniaudio не найден (нужна vcpkg-функция audio): звуковой движок не собирается")
endif()
set_property(TARGET han_audio PROPERTY SOURCES ${han_audio_sources})

# Демо звука на настоящем устройстве (E7): тон облетает слушателя. Нужен miniaudio.
if(han_audio_have_ma)
    add_executable(han_audio_demo apps/audio_demo.cpp)
    target_link_libraries(han_audio_demo PRIVATE han_audio)
    if(MSVC)
        target_compile_options(han_audio_demo PRIVATE /W4 /permissive- /utf-8)
    endif()
endif()
